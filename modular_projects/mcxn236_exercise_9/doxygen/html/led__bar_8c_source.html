84static bool is_power_of_two_or_edge(uint8_t mask, uint8_t num_leds)
85{
86 uint8_t ones = POPCOUNT(mask);
87
88 return (ones == 1) || (ones == (num_leds - 1));
89}

97static int led_bar_write_mask(const struct device *dev)
98{
99 const struct led_bar_config *cfg = dev->config;
100 const struct led_bar_data *data = dev->data;
101
102 LOG_DBG("mask=0x%02X", data->led_mask);
103
104 for (int i = 0; i < cfg->num_leds; i++) {
105 int ret = gpio_pin_set_dt(&cfg->gpios[i],
106 (data->led_mask >> i) & 0x01);
107
108 if (ret < 0) {
109 LOG_ERR("Failed to write LED %d: %d", i, ret);
110 return ret;
111 }
112 }
113
114 return 0;
115}

129static int led_bar_reset_isolated_bit(const struct device *dev)
130{
131 const struct led_bar_config *cfg = dev->config;
132 struct led_bar_data *data = dev->data;
133
134 uint8_t isolated_bit = (data->direction == DIR_RIGHT)
135 ? 0x01
136 : BIT(cfg->num_leds - 1);
137
138 if (data->polarity == INVERSE) {
139 data->led_mask = (uint8_t)~isolated_bit;
140 } else {
141 data->led_mask = isolated_bit;
142 }
143
144 return led_bar_write_mask(dev);
145}

184static int led_bar_rotate_impl(const struct device *dev)
185{
186 const struct led_bar_config *cfg = dev->config;
187 struct led_bar_data *data = dev->data;
188
189 if (!is_power_of_two_or_edge(data->led_mask, cfg->num_leds)) {
190 int ret = led_bar_reset_isolated_bit(dev);
191
192 return (ret < 0) ? ret : -EINVAL;
193 }
194
195 if (data->direction == DIR_RIGHT) {
196 data->led_mask = (uint8_t)((data->led_mask << 1) |
197 (data->led_mask >> (cfg->num_leds - 1)));
198 } else {
199 data->led_mask = (uint8_t)((data->led_mask >> 1) |
200 (data->led_mask << (cfg->num_leds - 1)));
201 }
202
203 return led_bar_write_mask(dev);
204}

322static int led_bar_init(const struct device *dev)
323{
324 const struct led_bar_config *cfg = dev->config;
325 struct led_bar_data *data = dev->data;
326
327 if (cfg->num_leds < LED_BAR_MIN_LEDS ||
328 cfg->num_leds > LED_BAR_MAX_LEDS) {
329 LOG_ERR("Unsupported LED count %u (must be %u..%u, "
330 "LED_BAR_MIN_LEDS..LED_BAR_MAX_LEDS)",
331 cfg->num_leds, LED_BAR_MIN_LEDS, LED_BAR_MAX_LEDS);
332 return -EINVAL;
333 }
334
335 data->led_mask = 0x00;
336 data->direction = DIR_RIGHT;
337 data->polarity = DIRECT;
338
339 for (int i = 0; i < cfg->num_leds; i++) {
340 if (!gpio_is_ready_dt(&cfg->gpios[i])) {
341 LOG_ERR("LED bar GPIO %d device is not ready", i);
342 return -ENODEV;
343 }
344
345 int ret = gpio_pin_configure_dt(&cfg->gpios[i],
346 GPIO_OUTPUT_INACTIVE);
347
348 if (ret < 0) {
349 LOG_ERR("Failed to configure LED bar GPIO %d: %d",
350 i, ret);
351 return ret;
352 }
353 }
354
355 return 0;
356}
